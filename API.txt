//Запросы авторизации:

//Запрос на логин пользователя
POST /auth/login 
Body: {login: string, password: string} 
Response: {
	tokens:{refreshToken: string, accessToken: string}, 
	user: { id: string, login: string, username: string, role: string}
}

//Запрос на регистрацию пользователя
POST /auth/register
Body: {login: string, password: string, username: string} 
Response: {
	tokens:{refreshToken: string, accessToken: string}, 
	user: { id: string, login: string, username: string, role: string}
}

//Запрос на обновления токена jwt
POST /auth/refresh
Body: {refreshToken: string},
Response: {refreshToken: string, accessToken: string}

//Запрос на разлогин пользователя
GET /auth/logout
Headers: {authorization: Bearer token}
Response: void

//Запрос на авторизацию пользователя по токену jwt
GET /auth
Headers: {authorization: Bearer token}
Response: {id: string, login: string, username: string, role: string}

//Запросы контента:

//Запрос на список постов
GET /content
Query: {page: number, limit: number}
Response: {
	data: [{id: string, author: { id: string, login: string, username: string, role: string }, text: string likeCount: number, repliesCount: number}],
	meta: { totalItems: number, itemCount: number, itemsPerPage: number, totalPages: number, currentPage: number}
}

//Запрос постов пользователя

Query: {page: number, limit: number}

	data: [{id: string, author: { id: string, login: string, username: string, role: string }, text: string likeCount: number, repliesCount: number}],
	meta: { totalItems: number, itemCount: number, itemsPerPage: number, totalPages: number, currentPage: number}
}
